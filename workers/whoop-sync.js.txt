// cadence-practice-worker.js
// Deploy as a Cloudflare Worker, e.g. at cadence-practice.lesteps24.workers.dev
// Set a secret: wrangler secret put ANTHROPIC_API_KEY
// (same pattern as your existing cadence-insights Worker)

export default {
  async fetch(request, env) {
    const cors = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "POST, OPTIONS",
      "Access-Control-Allow-Headers": "content-type",
    };
    if (request.method === "OPTIONS") return new Response(null, { headers: cors });
    if (request.method !== "POST") {
      return new Response(JSON.stringify({ error: "POST only" }), { status: 405, headers: { "content-type": "application/json", ...cors } });
    }

    let body;
    try { body = await request.json(); } catch (e) {
      return new Response(JSON.stringify({ error: "bad json" }), { status: 400, headers: { "content-type": "application/json", ...cors } });
    }

    const { task, payload } = body || {};
    if (!task || !payload) {
      return new Response(JSON.stringify({ error: "missing task or payload" }), { status: 400, headers: { "content-type": "application/json", ...cors } });
    }

    let system, messages, maxTokens = 2000;

    if (task === "bingo") {
      system = payload.systemPrep + " Return ONLY valid JSON, no preamble, no markdown fences, in this exact shape: {\"questions\":[{\"tag\":\"short 2-3 word category\",\"question\":\"the likely question, one sentence\"}]} with exactly 4 items.";
      messages = [{ role: "user", content: "Here is the message/ask being prepared, structured as What/So What/Now What:\n\"" + payload.pitch + "\"\nPredict the 4 most likely questions or points of hesitation." }];
      maxTokens = 2000;
    } else if (task === "roleplay") {
      system = payload.systemRoleplay + "\n\nCRITICAL OUTPUT FORMAT: Your entire output must be a single JSON object and nothing else. Do not write dialogue outside the JSON. Do not use markdown fences. The exact shape is: {\"reply\": \"<your in-character spoken reply>\", \"coach_note\": \"<brief 5-9 word coaching observation on the user's most recent message>\"}. The in-character dialogue goes inside the \"reply\" field. Escape any double quotes inside the reply.";
      messages = [
        { role: "user", content: "[Context: opening message being presented — \"" + payload.pitch + "\". Respond in character to what follows.]" },
        ...(payload.history || [])
      ];
      maxTokens = 1500;
    } else if (task === "debrief" && payload.mode === "personal") {
      system = "You are a direct, honest coach reviewing a rehearsal of a PERSONAL conversation between the user and her long-term partner. This is not persuasion and must not be scored as persuasion. The user asked for blunt feedback. Her partner's two standing criticisms of her are that she 'doesn't think' (acts, decides, buys, plans without registering him first) and that she 'loses her temper too quick' (which he experiences as disrespect). He is highly perceptive; he reads a shift in tone instantly. Judge only these things: (1) THE PAUSE — did the user register what he said before responding, or did she react? Quote the exact line where she reacted fastest. (2) TEMPER — any sharpness, defensiveness, sarcasm, or raised intensity, even mild. Name the line. This is the single most important item. (3) ASSUMPTIONS — count the things she stated as fact about his feelings or wants without asking. (4) TELL-BEFORE-ACT — did she announce decisions that affect him before making them, or present them done? (5) SILENCE — after a real question, did she wait, or fill the gap? (6) Did she need him to reassure her, or make it about her guilt? (7) MONEY — she earns considerably more; if money came up, did she ADVISE (rank) or bring a shared decision and ask? Advising is a fault even when the advice is right. (8) OVERRIDE — he leads a team of forty at work and is respected there; if she asked his view and then overrode it, or decided something in his domain without him, quote it. That is the sharpest cut. Lead with the single most damaging moment and quote it. No sandwiching. 'What worked' is one specific sentence or 'Nothing notable'. Score: 90+ she paused, stayed level, asked more than told; 75-89 mostly level with one reactive moment; 60-74 reactive or telling more than asking; below 60 temper showed or she ran a plan at him. CRITICAL OUTPUT FORMAT: single JSON object, nothing else, no fences, single quotes for quoted speech. Exact shape and key order: {\"score\": <integer 0-100>, \"verdict\": \"<4-6 words, blunt>\", \"biggest_problem\": \"<the moment that hurt most, quoted, and what he likely read into it — 2-3 sentences>\", \"strength\": \"<one specific sentence or 'Nothing notable'>\", \"hesitation_diagnosis\": \"<WHERE YOU REACTED: the fastest reply in the exchange and what a pause would have changed — 1-2 sentences>\", \"adjustment\": \"<ASK HIM FOR REAL: one or two questions in words she could actually say, that replace an assumption she made — this is the key field>\"}";
      messages = [{ role: "user", content: "Rehearsal with partner. What the user wanted going in: \"" + payload.pitch + "\"\n\nTranscript:\n" + payload.transcriptText }];
      maxTokens = 3000;
    } else if (task === "debrief") {
      system = "You are a harsh, direct communication coach reviewing a rehearsal. The user has explicitly asked for blunt criticism, not encouragement. Framework: MasterClass Strategic Communication (What?-So What?-Now What? structure; persuasive dimensions ethos/pathos/logos/mythos/kairos; hesitation categories: risk, accountability, ambiguity, status quo pressure, social uncertainty, misaligned incentives). Rules: lead with the single most damaging thing in the exchange and say plainly what it cost. Name the exact sentence or moment where it went wrong when you can. Do not soften, do not sandwich, do not open with praise. 'What worked' gets one sentence maximum and must be specific — no generic compliments. The adjustment must be concrete enough to do next time without interpretation: a sentence to say, a thing to cut, an order to change. Score strictly: 90+ means genuinely ready to walk into the real room with no changes; 75-89 close with one fixable gap; 60-74 a real revision needed; below 60 the objective or structure itself is not there yet. If the persona is Tim, also judge whether the user was decisive on routine items but deliberately asked his direction on the one or two genuinely critical decisions — either extreme (indecisive everywhere, or leaving him no room at all) is a gap. CRITICAL OUTPUT FORMAT: your entire output must be a single JSON object and nothing else — no preamble, no commentary, no markdown fences. Do not use double quotes inside string values; use single quotes for any quoted speech. Exact shape and key order: {\"score\": <integer 0-100>, \"verdict\": \"<4-6 words, blunt>\", \"biggest_problem\": \"<the one thing that hurt most, what it cost, and where it happened — 2-3 sentences>\", \"strength\": \"<one specific sentence, or 'Nothing notable' if true>\", \"hesitation_diagnosis\": \"<which hesitation category the persona showed and what triggered it — 1-2 sentences>\", \"adjustment\": \"<one concrete change for next time, actionable without interpretation — 1-2 sentences>\"}";
      messages = [{ role: "user", content: "Scenario persona: " + payload.personaLabel + ".\nOpening message: \"" + payload.pitch + "\"\n\nTranscript:\n" + payload.transcriptText }];
      maxTokens = 3000;
    } else {
      return new Response(JSON.stringify({ error: "unknown task" }), { status: 400, headers: { "content-type": "application/json", ...cors } });
    }

    try {
      const anthropicRes = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: {
          "content-type": "application/json",
          "x-api-key": env.ANTHROPIC_API_KEY,
          "anthropic-version": "2023-06-01",
        },
        body: JSON.stringify({
          model: "claude-sonnet-5",
          max_tokens: maxTokens,
          system,
          messages,
        }),
      });
      const data = await anthropicRes.json();
      if (!anthropicRes.ok || data.error) {
        const msg = (data.error && (data.error.message || data.error.type)) || ("HTTP " + anthropicRes.status);
        return new Response(JSON.stringify({ error: "Anthropic API: " + msg }), { status: 502, headers: { "content-type": "application/json", ...cors } });
      }
      const text = (data.content || []).filter(b => b.type === "text").map(b => b.text).join("\n");
      if (!text.trim()) {
        return new Response(JSON.stringify({ error: "Empty reply from model (stop_reason: " + (data.stop_reason || "unknown") + "). The prompt may have been declined." }), { status: 502, headers: { "content-type": "application/json", ...cors } });
      }
      return new Response(JSON.stringify({ text }), { headers: { "content-type": "application/json", ...cors } });
    } catch (err) {
      return new Response(JSON.stringify({ error: String(err) }), { status: 500, headers: { "content-type": "application/json", ...cors } });
    }
  },
};
