/**
 * Whoop v2 sync worker (Cloudflare).
 * Mirrors the shape of your oura-worker so the app can consume it the same way:
 *   GET  /            -> { days: [ { day, sleepHours, deep, rem, efficiency, latency,
 *                                    breath, sleepScore, recovery, hrv, rhr, spo2, strain }, ... ] }
 *   GET  /?days=21    -> same, custom window (default 21, capped 60)
 *   GET  /auth        -> one-time: redirects you to Whoop consent
 *   GET  /callback    -> Whoop redirects here; stores your refresh token, shows "connected"
 *   GET  /health      -> quick status (are we connected?)
 *
 * Whoop hands back a NEW refresh token on every refresh and kills the old one, so we persist
 * it to KV (binding: WHOOP_KV) every cycle. The token never touches the browser.
 *
 * Bindings / vars to set in the Cloudflare dashboard:
 *   - KV namespace binding:  WHOOP_KV
 *   - Secret:  WHOOP_CLIENT_ID
 *   - Secret:  WHOOP_CLIENT_SECRET
 * Redirect URI to register in the Whoop dashboard:  https://<this-worker>/callback
 */

const AUTH_URL = "https://api.prod.whoop.com/oauth/oauth2/auth";
const TOKEN_URL = "https://api.prod.whoop.com/oauth/oauth2/token";
const API = "https://api.prod.whoop.com/developer";
const SCOPES = "offline read:sleep read:recovery read:cycles";
const KV_KEY = "whoop_refresh_token";

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type",
};

function json(obj, status = 200) {
  return new Response(JSON.stringify(obj), { status, headers: { "Content-Type": "application/json", ...CORS } });
}
function html(body, status = 200) {
  return new Response(`<!doctype html><meta name=viewport content="width=device-width,initial-scale=1"><body style="font-family:system-ui;max-width:560px;margin:40px auto;padding:0 16px;line-height:1.5;color:#211D17">${body}</body>`, { status, headers: { "Content-Type": "text/html" } });
}

// ISO timestamp + offset like "-04:00" -> local "YYYY-MM-DD"
function localDate(iso, offset) {
  if (!iso) return null;
  let ms = Date.parse(iso);
  if (isNaN(ms)) return null;
  if (offset && /^[+-]\d{2}:\d{2}$/.test(offset)) {
    const sign = offset[0] === "-" ? -1 : 1;
    const mins = parseInt(offset.slice(1, 3), 10) * 60 + parseInt(offset.slice(4, 6), 10);
    ms += sign * mins * 60000;
  }
  return new Date(ms).toISOString().slice(0, 10);
}
const round1 = (n) => (n == null ? null : Math.round(n * 10) / 10);
const msToHr = (ms) => (ms == null ? null : round1(ms / 3600000));
const msToMin = (ms) => (ms == null ? null : Math.round(ms / 60000));

async function refreshAccessToken(env) {
  const stored = await env.WHOOP_KV.get(KV_KEY);
  if (!stored) throw new Error("not_connected");
  const body = new URLSearchParams({
    grant_type: "refresh_token",
    refresh_token: stored,
    client_id: env.WHOOP_CLIENT_ID,
    client_secret: env.WHOOP_CLIENT_SECRET,
    scope: "offline",
  });
  const r = await fetch(TOKEN_URL, { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body });
  if (!r.ok) throw new Error("refresh_failed_" + r.status);
  const t = await r.json();
  // Whoop rotates the refresh token — persist the new one immediately or the next sync breaks.
  if (t.refresh_token) await env.WHOOP_KV.put(KV_KEY, t.refresh_token);
  return t.access_token;
}

async function pagedGet(path, token, startISO, endISO) {
  const out = [];
  let nextToken = null;
  for (let guard = 0; guard < 6; guard++) {
    const u = new URL(API + path);
    u.searchParams.set("limit", "25");
    if (startISO) u.searchParams.set("start", startISO);
    if (endISO) u.searchParams.set("end", endISO);
    if (nextToken) u.searchParams.set("nextToken", nextToken);
    const r = await fetch(u.toString(), { headers: { Authorization: "Bearer " + token } });
    if (!r.ok) throw new Error("api_" + r.status + "_" + path);
    const j = await r.json();
    (j.records || []).forEach((rec) => out.push(rec));
    nextToken = j.next_token || null;
    if (!nextToken) break;
  }
  return out;
}

async function buildDays(env, days) {
  const token = await refreshAccessToken(env);
  const end = new Date();
  const start = new Date(end.getTime() - days * 86400000);
  const startISO = start.toISOString();
  const endISO = end.toISOString();

  const safeGet = async (path) => {
    try { return { data: await pagedGet(path, token, startISO, endISO) }; }
    catch (e) { return { data: [], error: String(e && e.message || e) }; }
  };
  const [sl, rc, cy] = await Promise.all([
    safeGet("/v2/activity/sleep"),
    safeGet("/v2/recovery"),
    safeGet("/v2/cycle"),
  ]);
  const warnings = [];
  if (sl.error) warnings.push("sleep: " + sl.error);
  if (rc.error) warnings.push("recovery: " + rc.error);
  if (cy.error) warnings.push("cycle/strain: " + cy.error);
  const sleeps = sl.data, recoveries = rc.data, cycles = cy.data;

  const map = {}; // day -> record
  const dayOf = (d) => (map[d] = map[d] || { day: d });
  const sleepIdToDay = {};

  // main night sleeps (skip naps)
  sleeps.forEach((s) => {
    if (s.nap) return;
    const day = localDate(s.end, s.timezone_offset);
    if (!day) return;
    sleepIdToDay[s.id] = day;
    const d = dayOf(day);
    const sc = s.score || {};
    const ss = sc.stage_summary || {};
    const inBed = ss.total_in_bed_time_milli;
    const awake = ss.total_awake_time_milli || 0;
    if (inBed != null) d.sleepHours = msToHr(inBed - awake);
    if (inBed != null) d.inBedHours = msToHr(inBed);
    d.deep = msToMin(ss.total_slow_wave_sleep_time_milli);
    d.rem = msToMin(ss.total_rem_sleep_time_milli);
    d.light = msToMin(ss.total_light_sleep_time_milli);
    if (awake) d.awakeMin = msToMin(awake);
    if (ss.disturbance_count != null) d.disturbances = ss.disturbance_count;
    if (ss.sleep_cycle_count != null) d.cycles = ss.sleep_cycle_count;
    if (sc.sleep_efficiency_percentage != null) d.efficiency = round1(sc.sleep_efficiency_percentage);
    if (sc.sleep_performance_percentage != null) d.sleepScore = Math.round(sc.sleep_performance_percentage);
    if (sc.sleep_consistency_percentage != null) d.consistency = Math.round(sc.sleep_consistency_percentage);
    if (sc.respiratory_rate != null) d.breath = round1(sc.respiratory_rate);
  });

  // recovery (recovery_score -> readiness slot; plus hrv/rhr/spo2)
  recoveries.forEach((rec) => {
    const day = sleepIdToDay[rec.sleep_id] || localDate(rec.created_at, null);
    if (!day) return;
    const d = dayOf(day);
    const sc = rec.score || {};
    if (sc.recovery_score != null) d.recovery = Math.round(sc.recovery_score);
    if (sc.hrv_rmssd_milli != null) { let h = sc.hrv_rmssd_milli; if (h < 1) h = h * 1000; d.hrv = Math.round(h); } // normalize seconds->ms so it lines up with Oura/Withings
    if (sc.resting_heart_rate != null) d.rhr = Math.round(sc.resting_heart_rate);
    if (sc.spo2_percentage != null) d.spo2 = round1(sc.spo2_percentage);
  });

  // strain (cycle.score.strain, 0-21)
  cycles.forEach((c) => {
    const day = localDate(c.start, c.timezone_offset);
    if (!day) return;
    const d = dayOf(day);
    const sc = c.score || {};
    if (sc.strain != null) d.strain = round1(sc.strain);
  });

  return { days: Object.values(map).sort((a, b) => (a.day < b.day ? 1 : -1)), warnings }; // newest first
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname.replace(/\/$/, "") || "/";
    if (request.method === "OPTIONS") return new Response(null, { headers: CORS });
    const redirectUri = url.origin + "/callback";

    try {
      if (path === "/auth") {
        const a = new URL(AUTH_URL);
        a.searchParams.set("response_type", "code");
        a.searchParams.set("client_id", env.WHOOP_CLIENT_ID);
        a.searchParams.set("redirect_uri", redirectUri);
        a.searchParams.set("scope", SCOPES);
        a.searchParams.set("state", crypto.randomUUID());
        return Response.redirect(a.toString(), 302);
      }

      if (path === "/callback") {
        const err = url.searchParams.get("error");
        if (err) return html("<h2>Whoop returned an error</h2><p style='font-family:monospace;background:#EFE7D6;padding:10px 12px;border-radius:8px'><b>" + err + "</b><br>" + (url.searchParams.get("error_description") || "") + "</p><p>Start over at <a href='/auth'>/auth</a>.</p>", 400);
        const code = url.searchParams.get("code");
        if (!code) return html("<h2>Missing code</h2><p>No <code>code</code> or <code>error</code> came back from Whoop. Start fresh at <a href='/auth'>/auth</a> and approve without using back/refresh.</p>", 400);
        const body = new URLSearchParams({
          grant_type: "authorization_code",
          code,
          client_id: env.WHOOP_CLIENT_ID,
          client_secret: env.WHOOP_CLIENT_SECRET,
          redirect_uri: redirectUri,
        });
        const r = await fetch(TOKEN_URL, { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body });
        if (!r.ok) return html("<h2>Token exchange failed</h2><p>Status " + r.status + ". Check client id/secret and redirect URI.</p>", 500);
        const t = await r.json();
        if (!t.refresh_token) return html("<h2>No refresh token returned</h2><p>Make sure the <code>offline</code> scope is included.</p>", 500);
        await env.WHOOP_KV.put(KV_KEY, t.refresh_token);
        return html("<h2>✓ Whoop connected</h2><p>You can close this tab. Le Cadence will start pulling Whoop as a fallback source.</p>");
      }

      if (path === "/setup") {
        return html(
          "<h2>Le Cadence · Whoop setup</h2>" +
          "<p>Register this <b>exact</b> redirect URI in your Whoop app (Developer Dashboard → your app → Redirect URIs). Copy it verbatim — no trailing slash, no spaces:</p>" +
          "<p style='font-family:monospace;font-size:15px;background:#EFE7D6;padding:12px 14px;border-radius:8px;word-break:break-all;user-select:all'>" + redirectUri + "</p>" +
          "<p>Once it's saved in Whoop, tap below to connect:</p>" +
          "<p><a href='/auth' style='display:inline-block;background:#1E2A4A;color:#fff;text-decoration:none;padding:10px 18px;border-radius:8px'>Connect Whoop →</a></p>" +
          "<p class='note' style='color:#6b6253;font-size:13px'>If connecting still fails, the URI above and the one in Whoop don't match character-for-character — that's the whole problem.</p>"
        );
      }

      if (path === "/health") {
        const has = !!(await env.WHOOP_KV.get(KV_KEY));
        return json({ connected: has, redirect_uri: redirectUri, scopes: SCOPES });
      }

      // default: data
      let days = parseInt(url.searchParams.get("days") || "21", 10);
      if (isNaN(days) || days < 1) days = 21;
      if (days > 60) days = 60;
      const result = await buildDays(env, days);
      return json({ days: result.days, count: result.days.length, warnings: result.warnings, source: "whoop" });
    } catch (e) {
      const msg = String(e && e.message || e);
      if (msg === "not_connected") return json({ error: "not_connected", hint: "Visit /auth once to connect Whoop." }, 401);
      if (msg.startsWith("refresh_failed")) return json({ error: msg, hint: "Refresh token expired or revoked — visit /auth again." }, 401);
      return json({ error: msg }, 500);
    }
  },
};
