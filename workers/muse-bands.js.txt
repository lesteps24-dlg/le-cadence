// muse-bands — Cloudflare Worker
// Reads the five EEG band values (Delta/Theta/Alpha/Beta/Gamma) off a Muse / Enso screenshot,
// so you never have to type them. Your existing muse-analyze Worker is left untouched.
//
// POST { imageBase64, imageType }
//   -> { powerbands: { delta, theta, alpha, beta, gamma } }   (values in dB; null if unreadable)
//
// ── Deploy ───────────────────────────────────────────────────────────────────
// 1. Workers & Pages → Create → Worker → name it "muse-bands" → paste this file.
// 2. Settings → Variables and Secrets → add SECRET  ANTHROPIC_API_KEY  (your key).
// 3. Deploy, copy the URL (e.g. https://muse-bands.<you>.workers.dev/).
// 4. In index.html set  MUSE_BANDS_URL  (top of the Muse app) to that URL. Redeploy the page.
// ──────────────────────────────────────────────────────────────────────────────

const MODEL = "claude-sonnet-4-6"; // vision-capable
const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Access-Control-Allow-Headers": "content-type",
};
function json(b, s) {
  return new Response(JSON.stringify(b), { status: s || 200, headers: { "content-type": "application/json", ...CORS } });
}

const SYSTEM = `You read brain-power readouts from Muse / Enso meditation screenshots. The image lists five EEG frequency bands — Delta, Theta, Alpha, Beta, Gamma — each with a numeric value, usually in dB (values may be negative). Extract each band's value exactly as shown. Return ONLY strict minified JSON, no prose and no code fence: {"delta":<number|null>,"theta":<number|null>,"alpha":<number|null>,"beta":<number|null>,"gamma":<number|null>}. Use null for any band you cannot clearly read. Never invent or estimate a value.`;

export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") return new Response(null, { headers: CORS });
    if (request.method !== "POST") return json({ error: "POST only" }, 405);
    if (!env.ANTHROPIC_API_KEY) return json({ error: "ANTHROPIC_API_KEY secret is not set on the Worker" }, 500);

    let body;
    try { body = await request.json(); } catch (e) { return json({ error: "invalid JSON body" }, 400); }
    const data64 = body.imageBase64 || body.chartImage;
    const mediaType = body.imageType || body.chartImageType || "image/png";
    if (!data64) return json({ error: "no image provided" }, 400);

    try {
      const r = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "content-type": "application/json", "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" },
        body: JSON.stringify({
          model: MODEL,
          max_tokens: 300,
          system: SYSTEM,
          messages: [{
            role: "user",
            content: [
              { type: "image", source: { type: "base64", media_type: mediaType, data: data64 } },
              { type: "text", text: "Read the five band values from this screenshot and return the JSON." },
            ],
          }],
        }),
      });
      const d = await r.json();
      if (!r.ok) return json({ error: "anthropic request failed", detail: JSON.stringify(d) }, 502);

      const text = (d.content || []).filter((b) => b.type === "text").map((b) => b.text).join("").trim();
      let pb;
      try { pb = JSON.parse(text.replace(/```json|```/g, "").trim()); }
      catch (e) { return json({ error: "could not parse band values from the image", detail: text.slice(0, 200) }, 502); }

      const num = (v) => (v == null || v === "" || isNaN(Number(v)) ? null : Number(v));
      return json({ powerbands: { delta: num(pb.delta), theta: num(pb.theta), alpha: num(pb.alpha), beta: num(pb.beta), gamma: num(pb.gamma) } });
    } catch (e) {
      return json({ error: "worker error", detail: String((e && e.message) || e) }, 500);
    }
  },
};
