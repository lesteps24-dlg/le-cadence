/* ============================================================
   THE HUNDRED DAYS — trends analyzer (Cloudflare Worker)
   ============================================================

   This little Worker is the ONLY place your Anthropic key lives.
   The habit-tracker app sends it your notes; it asks Claude for
   the trends and sends the result back. The key is never exposed
   to the browser.

   DEPLOY (dashboard, ~3 min):
   1) Cloudflare dashboard -> Workers & Pages -> Create -> Worker.
   2) Name it (e.g. hundred-days-analyze). Click Deploy once to create it,
      then "Edit code", paste this whole file in, and Deploy again.
   3) Add your key as a SECRET:
        Worker -> Settings -> Variables and Secrets -> Add
        Name:  ANTHROPIC_API_KEY
        Type:  Secret (encrypted)
        Value: your Anthropic key (sk-ant-...)
      Deploy once more so the secret takes effect.
   4) Copy the Worker's URL (something.workers.dev) and paste it into
      ANALYZE_URL near the top of index.html. Re-deploy the app.

   That's it. Your key stays here, encrypted, server-side only.
   ============================================================ */

// Current Claude model. claude-sonnet-4-6 = good balance.
// Swap to "claude-haiku-4-5-20251001" for cheaper/faster.
// Verify current strings at https://docs.claude.com/en/docs/about-claude/models/overview
const MODEL = "claude-sonnet-4-6";

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type",
};

const SYSTEM =
  "You are a blunt, no-excuses accountability coach reviewing someone's 100-day habit challenge. " +
  "Tell the truth plainly; do not flatter and do not cushion. Name what is slipping and call out broken " +
  "intentions directly. You can be pointed, but stay constructive — blunt, not cruel, and never sarcastic. " +
  "You receive THREE kinds of data and must weigh all three: (1) per-habit consistency (days done out of the " +
  "challenge, current and best streaks), (2) body metrics tracked over time, and (3) optional written " +
  "reflections — which are frequently sparse or entirely absent. When they are, judge from the habit " +
  "consistency and the metric movement and do NOT say there is nothing to analyze. Ground every point in the " +
  "actual data: which habits are holding versus the ones she keeps missing, how the numbers are actually " +
  "moving, and where the reflections admit the same excuse more than once. " +
  "HARD LIMITS: be brutally honest about CONSISTENCY and EFFORT — never about her body, weight, or worth. Do " +
  "not shame, moralize about appearance, or push extreme, crash, or restrictive measures. Relate a metric to a " +
  "habit only as a possible association, never as proven cause, and never as a medical claim or diagnosis. " +
  "Plain English, no jargon, no hedging. Respond ONLY with the requested JSON, no markdown, no preamble.";

export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") return new Response(null, { headers: CORS });
    if (request.method !== "POST") return json({ error: "POST only" }, 405);
    if (!env.ANTHROPIC_API_KEY) return json({ error: "Worker is missing the ANTHROPIC_API_KEY secret" }, 500);

    try {
      const { notes = [], stats = {} } = await request.json();
      const hasNotes = Array.isArray(notes) && notes.length > 0;
      const hasHabits = Array.isArray(stats.perHabit) && stats.perHabit.length > 0;
      const hasMetrics = Array.isArray(stats.metrics) && stats.metrics.length >= 2;
      const hasNutrition = stats.nutrition && stats.nutrition.daysLogged > 0;
      if (!hasNotes && !hasHabits && !hasMetrics && !hasNutrition) {
        return json({ error: "Nothing to analyze yet — log some habits, body metrics, nutrition, or reflections first." }, 400);
      }

      const r = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: {
          "x-api-key": env.ANTHROPIC_API_KEY,
          "anthropic-version": "2023-06-01",
          "content-type": "application/json",
        },
        body: JSON.stringify({
          model: MODEL,
          max_tokens: 1024,
          system: SYSTEM,
          messages: [{ role: "user", content: buildPrompt(notes, stats) }],
        }),
      });

      if (!r.ok) return json({ error: "Claude API error", detail: await r.text() }, 502);

      const data = await r.json();
      const text = (data.content || [])
        .filter((b) => b.type === "text")
        .map((b) => b.text)
        .join("\n")
        .trim();

      let parsed = null;
      try { parsed = JSON.parse(text.replace(/```json|```/g, "").trim()); } catch (e) { /* fall back */ }

      return json(parsed ? { analysis: parsed } : { analysis: { summary: text } });
    } catch (e) {
      return json({ error: String(e) }, 500);
    }
  },
};

function metricMovement(metrics) {
  if (!Array.isArray(metrics) || metrics.length < 2) return "";
  const firstLast = (getter) => {
    let f = null, l = null;
    for (const m of metrics) { const v = parseFloat(getter(m)); if (!isNaN(v)) { if (f === null) f = v; l = v; } }
    return (f === null || l === null) ? null : [f, l];
  };
  const keys = [
    ["weight", "Weight"], ["bodyFat", "Body fat %"], ["hrv", "HRV"], ["rhr", "RHR"],
    ["systolic", "BP systolic"], ["diastolic", "BP diastolic"], ["energy", "Energy"],
  ];
  const lines = [];
  for (const [k, label] of keys) {
    const fl = firstLast((m) => m[k]);
    if (fl) { const d = +(fl[1] - fl[0]).toFixed(1); lines.push(`  ${label}: ${fl[0]} -> ${fl[1]} (${d > 0 ? "+" : ""}${d})`); }
  }
  const pp = firstLast((m) => (m.systolic && m.diastolic) ? (Number(m.systolic) - Number(m.diastolic)) : "");
  if (pp) { const d = +(pp[1] - pp[0]).toFixed(1); lines.push(`  Pulse pressure: ${pp[0]} -> ${pp[1]} (${d > 0 ? "+" : ""}${d})`); }
  // also surface any tracked body measurements (waist, hips, etc.)
  const measKeys = new Set();
  metrics.forEach((m) => Object.keys(m.measurements || {}).forEach((k) => measKeys.add(k)));
  measKeys.forEach((mk) => {
    const fl = firstLast((m) => (m.measurements || {})[mk]);
    if (fl) { const d = +(fl[1] - fl[0]).toFixed(1); lines.push(`  ${mk}: ${fl[0]} -> ${fl[1]} (${d > 0 ? "+" : ""}${d})`); }
  });
  return lines.join("\n");
}

function buildPrompt(notes, stats) {
  const hasNotes = Array.isArray(notes) && notes.length > 0;
  const entries = (notes || [])
    .map((n) => `Date ${n.date}\n  Worked: ${n.worked || "—"}\n  Didn't: ${n.didnt || "—"}\n  Back on track: ${n.back || "—"}`)
    .join("\n\n");

  const perHabit = (stats.perHabit || [])
    .map((h) => `  ${h.name}: ${h.done}/${h.of || 100} days done (streak ${h.streak || 0}, best ${h.best || 0})`)
    .join("\n");

  const movement = metricMovement(stats.metrics);

  const n = stats.nutrition;
  const nutrition = n ? (
    "  Days logged: " + n.daysLogged + "\n" +
    "  Fasting: avg " + (n.avgFastHrs != null ? n.avgFastHrs + "h" : "—") + " over " + (n.fastsLogged || 0) + " logged fasts; hit the 16:8 window (>=16h) on " + (n.hit16of8 || 0) + " of them\n" +
    "  Whole foods: " + (n.wholeFoodMealRate || "0%") + " of meals were whole-food; " + (n.allWholeFoodDays || 0) + " fully whole-food days\n" +
    "  Snacks: avg " + (n.avgSnacksPerDay != null ? n.avgSnacksPerDay + "/day" : "—") + "; processed sugar on " + (n.processedSugarDays || 0) + " days"
  ) : "";

  const metrics = (stats.metrics || [])
    .map((m) => {
      const meas = Object.entries(m.measurements || {})
        .filter(([, v]) => v !== "" && v != null)
        .map(([k, v]) => `${k} ${v}`)
        .join(", ");
      const parts = [
        m.weight ? `weight ${m.weight}` : "",
        m.bodyFat ? `body fat ${m.bodyFat}%` : "",
        m.hrv ? `HRV ${m.hrv}` : "",
        m.rhr ? `RHR ${m.rhr}` : "",
        (m.systolic && m.diastolic) ? `BP ${m.systolic}/${m.diastolic} (pulse pressure ${Number(m.systolic) - Number(m.diastolic)})` : "",
        m.energy ? `energy ${m.energy}` : "",
        meas,
      ].filter(Boolean).join(", ");
      return `  ${m.label}${m.date ? " (" + m.date + ")" : ""}: ${parts || "—"}`;
    })
    .join("\n");

  return (
    "My 100-day habit challenge data. Day " + (stats.dayNum || "?") + " of 100.\n" +
    "Analyze ALL of the data below. Habit consistency and metric movement are the primary " +
    "signal; reflections are supporting context and may be empty.\n\n" +
    "=== HABIT CONSISTENCY (each habit's progress to 100) ===\n" + (perHabit || "  (none yet)") + "\n\n" +
    "=== METRIC MOVEMENT (first -> latest entry) ===\n" + (movement || "  (need at least two metric checkpoints)") + "\n\n" +
    "=== METRIC DETAIL (every checkpoint) ===\n" + (metrics || "  (none logged yet)") + "\n\n" +
    (nutrition ? "=== NUTRITION & FASTING ===\n" + nutrition + "\n\n" : "") +
    "=== REFLECTIONS ===\n" + (hasNotes ? entries : "  (no written reflections in this window — analyze from the habits and metrics above; do not treat their absence as a lack of data)") + "\n\n" +
    "Read which habits are holding vs slipping by streak and completion. Where the metric " +
    "timeline has two or more points, comment on direction (for context: lower is generally " +
    "better for weight, body fat, RHR, and blood pressure; higher is generally better for HRV " +
    "and energy) and whether the movement plausibly tracks with how consistently I'm hitting " +
    "the related habits (e.g. sleep/exercise vs RHR or HRV). Note correlation, not proven " +
    "causation, and don't overclaim from thin data." +
    (nutrition ? " If nutrition data is present, judge fasting consistency against the 16:8 goal and whole-food rate bluntly, and note any plausible link between processed-sugar/snacking and the metric direction — association, not diagnosis." : "") + "\n\n" +
    'Return ONLY this JSON shape and nothing else:\n' +
    '{"summary":"a blunt 2-3 sentence verdict on how it is ACTUALLY going, drawn from habits and metrics — no hedging",' +
    '"signals":["how specific metrics are moving relative to habit consistency, stated plainly; [] only if fewer than two metric points exist"],' +
    '"themes":["recurring patterns in the reflections, including excuses that repeat; [] if there are no reflections"],' +
    '"working":["habits or behaviors that are genuinely holding — only ones the data supports"],' +
    '"obstacles":["the habits being missed and the excuses that recur — name them directly"],' +
    '"suggestions":["2-4 concrete, non-negotiable adjustments for the next stretch"]}'
  );
}

function json(obj, status = 200) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: { "content-type": "application/json", ...CORS },
  });
}
